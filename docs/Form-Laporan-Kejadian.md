    // Handle category selection
    function selectCategory(categoryName, element) {
      document.querySelectorAll('.category-btn').forEach(btn => {
        btn.classList.remove('bg-primary-container', 'text-on-primary', 'ring-2', 'ring-secondary');
        btn.classList.add('bg-surface-container-low', 'text-on-surface');
        const dot = btn.querySelector('.dot-indicator');
        if (dot) dot.classList.replace('bg-secondary-fixed', 'bg-transparent');
      });

      element.classList.remove('bg-surface-container-low', 'text-on-surface');
      element.classList.add('bg-primary-container', 'text-on-primary', 'ring-2', 'ring-secondary');
      const activeDot = element.querySelector('.dot-indicator');
      if (activeDot) activeDot.classList.replace('bg-transparent', 'bg-secondary-fixed');

      document.getElementById('kategoriInput').value = categoryName;
      document.getElementById('selectedCatLabel').textContent = categoryName;
    }

    // Set realtime date & time defaults
    function setRealtimeNow() {
      const now = new Date();
      const hours = String(now.getHours()).padStart(2, '0');
      const minutes = String(now.getMinutes()).padStart(2, '0');
      document.getElementById('jamKejadian').value = `${hours}:${minutes}`;
      
      const year = now.getFullYear();
      const month = String(now.getMonth() + 1).padStart(2, '0');
      const day = String(now.getDate()).padStart(2, '0');
      document.getElementById('tanggalKejadian').value = `${year}-${month}-${day}`;
    }

    // Initial setup for time inputs
    document.addEventListener('DOMContentLoaded', () => {
      setRealtimeNow();

      // Dynamic character counter for textarea
      const descArea = document.getElementById('deskripsiKejadian');
      const counter = document.getElementById('charCounter');
      if (descArea && counter) {
        descArea.addEventListener('input', () => {
          counter.textContent = `${descArea.value.length} Karakter`;
        });
      }

      // Map click simulation to move pin & update coordinates
      const mapBox = document.getElementById('mapInteractionBox');
      const pin = document.getElementById('interactivePin');
      const inputLat = document.getElementById('inputLat');
      const inputLng = document.getElementById('inputLng');
      const regionSpan = document.getElementById('detectedRegionName');

      if (mapBox && pin) {
        mapBox.addEventListener('click', (e) => {
          const rect = mapBox.getBoundingClientRect();
          const x = e.clientX - rect.left;
          const y = e.clientY - rect.top;

          pin.style.left = `${x}px`;
          pin.style.top = `${y}px`;

          // Generate simulated accurate coordinates in Thousand Islands / Java Sea
          const simulatedLat = (-5.900000 - (y / rect.height) * 0.15).toFixed(6);
          const simulatedLng = (106.650000 + (x / rect.width) * 0.15).toFixed(6);

          if (inputLat) inputLat.value = simulatedLat;
          if (inputLng) inputLng.value = simulatedLng;
          
          if (regionSpan) {
            regionSpan.textContent = "Kepulauan Seribu Bagian Selatan (Zona Konservasi Pemanfaatan Terbatas)";
          }
        });
      }
    });

    // Simulate GPS Current Location Trigger
    function captureCurrentGPS() {
      const inputLat = document.getElementById('inputLat');
      const inputLng = document.getElementById('inputLng');
      const pin = document.getElementById('interactivePin');
